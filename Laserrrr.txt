--// Ping display
local function updatePing()
    local success, ping = pcall(function()
        return Stats.Network.ServerStatsItem["Data Ping"]:GetValue()
    end)

    if success and ping then
        currentPing = math.floor(ping)

        if currentPing > 400 then
            PingLabel.TextColor3 = Color3.fromRGB(255, 0, 0)
        elseif currentPing > 300 then
            PingLabel.TextColor3 = Color3.fromRGB(255, 165, 0)
        elseif currentPing > 200 then
            PingLabel.TextColor3 = Color3.fromRGB(255, 255, 0)
        else
            PingLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
        end

        PingLabel.Text = "Ping: " .. currentPing .. " ms"
    else
        PingLabel.Text = "Ping: N/A"
    end
end

task.spawn(function()
    while task.wait(2) do
        updatePing()
    end
end)

updatePing()

--// Custom dragging
local dragging = false
local dragInput
local dragStart
local startPosition

local function makeDraggable(guiObject)
    guiObject.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPosition = MainFrame.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    guiObject.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement
            or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
end

makeDraggable(MainFrame)
makeDraggable(Title)
makeDraggable(MinimizedHeader)

UserInputService.InputChanged:Connect(function(input)
    if not dragging or input ~= dragInput then
        return
    end

    local delta = input.Position - dragStart
    MainFrame.Position = UDim2.new(
        startPosition.X.Scale,
        startPosition.X.Offset + delta.X,
        startPosition.Y.Scale,
        startPosition.Y.Offset + delta.Y
    )
end)

--// Laser gun firing
local function fireLaserGun()
    local Character = LocalPlayer.Character
    if not Character then
        return
    end

    local Tool = Character:FindFirstChildOfClass("Tool")
    if not Tool or Tool.Name ~= "Laser Gun" then
        return
    end

    local origin = Character.PrimaryPart and Character.PrimaryPart.Position
    if not origin then
        return
    end

    for _ = 1, 60 do
        local requestId = HttpService:GenerateGUID(false):lower():gsub("%-", "")
        local direction = Vector3.new(
            math.random(-2, 2),
            math.random(-2, 2),
            math.random(-2, 2)
        ).Unit

        LaserGunFire:FireServer(
            requestId,
            origin,
            direction,
            workspace:GetServerTimeNow()
        )

        local secondRequestId = HttpService:GenerateGUID(false):lower():gsub("%-", "")
        local offset = Vector3.new(
            math.random(-10, 10),
            math.random(-10, 10),
            math.random(-10, 10)
        )

        LaserGunFire:FireServer(
            secondRequestId,
            origin + offset,
            direction,
            workspace:GetServerTimeNow()
        )
    end
end

--// Automatic firing loop
 task.spawn(function()
    while true do
        if killerGunEnabled then
            local Character = LocalPlayer.Character

            if Character then
                local Tool = Character:FindFirstChildOfClass("Tool")

                if Tool and Tool.Name == "Laser Gun" then
                    fireLaserGun()
                    task.wait(0.1)
                end
            end
        end

        task.wait(0.01)
    end
end)
