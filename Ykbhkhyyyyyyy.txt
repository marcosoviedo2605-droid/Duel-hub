local function clamp(v, min, max)
    return math.max(min, math.min(max, v))
end

local function lerp(a, b, t)
    return a + (b - a) * t
end

local function cleanText(text)
    if not text then return "" end
    text = text:gsub("^%s+", ""):gsub("%s+$", "")
    return text
end

local function splitWords(text)
    local words = {}
    if text then
        for word in text:gmatch("%S+") do
            table.insert(words, word)
        end
    end
    return words
end

local function isRedeemable(text)
    if not text then return false end
    text = text:lower()
    return text:find("spawned") or text:find("redeemed") or text:find("prize") or text:find("reward")
end

local function isRedeemed(text)
    if not text then return false end
    text = text:lower()
    return text:find("redeemed")
end

local function formatUrl(str)
    return str:gsub("([^%w%-_%.~])", function(c)
        return string.format("%%%02X", string.byte(c))
    end)
end

local function hasFileFunctions()
    local ok, f = pcall(function() return readfile end)
    if ok and type(f) == "function" then return true end
    return false
end

local function saveState(data)
    if not hasFileFunctions() then return false end
    pcall(function()
        writefile("CloverRedeemerGui_state_AI.json", HttpService:JSONEncode(data))
    end)
    return true
end

local function loadState()
    if not hasFileFunctions() then return nil end
    local ok, data = pcall(function()
        if isfile("CloverRedeemerGui_state_AI.json") then
            return HttpService:JSONDecode(readfile("CloverRedeemerGui_state_AI.json"))
        end
    end)
    if ok and data then return data end
    return nil
end

-- ============================================
-- HTTP REQUEST HELPERS
-- ============================================

local function httpRequest(method, url, headers, body)
    local ok, result = pcall(function()
        if syn and syn.request then
            return syn.request({
                Method = method,
                Url = url,
                Headers = headers or {},
                Body = body or ""
            })
        elseif http and http.request then
            return http.request({
                Method = method,
                Url = url,
                Headers = headers or {},
                Body = body or ""
            })
        elseif http_request then
            return http_request({
                Method = method,
                Url = url,
                Headers = headers or {},
                Body = body or ""
            })
        else
            error("No HTTP request function available")
        end
    end)
    if not ok then return nil, result end
    return result
end

local function sendWebhook(webhookUrl, data)
    local headers = {
        ["Content-Type"] = "application/json"
    }
    local body = HttpService:JSONEncode(data)
    return httpRequest("POST", webhookUrl, headers, body)
end

local function applyTheme(frame)
    local theme = Theme.colors[Theme.current]
    for _, descendant in ipairs(frame:GetDescendants()) do
        if descendant:IsA("Frame") or descendant:IsA("ScrollingFrame") then
            if descendant.Name == "BG" then
                descendant.BackgroundColor3 = theme.BG
            elseif descendant.Name == "INNER" then
                descendant.BackgroundColor3 = theme.INNER
            elseif descendant.Name == "INNER_DARK" then
                descendant.BackgroundColor3 = theme.INNER_DARK
            end
        elseif descendant:IsA("TextLabel") or descendant:IsA("TextButton") or descendant:IsA("TextBox") then
            if descendant.Name == "TEXT" or descendant.Name == "Text" then
                descendant.TextColor3 = theme.TEXT
            end
            if descendant.Name == "DIM" then
                descendant.TextColor3 = theme.DIM
            end
            if descendant:GetAttribute("KeepScaleColor") then
                local keepColor = descendant:GetAttribute("KeepScaleColor")
                if type(keepColor) == "string" and Colors[keepColor] then
                    descendant.TextColor3 = Colors[keepColor]
                end
            end
        end
    end
    
    -- Apply theme strokes
    for _, stroke in ipairs(frame:GetDescendants()) do
        if stroke:IsA("UIStroke") and stroke:GetAttribute("ThemeStroke") then
            stroke.Color = theme.TEXT
        end
    end
end

local UIScale = Instance.new("UIScale")
UIScale.Name = "UIScale"
UIScale.Scale = 1

local function getScale()
    local viewport = Workspace.CurrentCamera.ViewportSize
    local scaleX = viewport.X / 1280
    local scaleY = viewport.Y / 720
    local baseScale = math.min(scaleX, scaleY)
    return clamp(baseScale * 1.1, 0.55, 1.35)
end

local function updateUIScale()
    local scale = getScale()
    if UIScale then
        UIScale.Scale = scale
    end
    return scale
end

    -- Set up content sizing
    local function updateContentSize()
        local totalHeight = 0
        for _, child in ipairs(ContentInner:GetChildren()) do
            if child:IsA("Frame") and child.Name == "Panels" then
                totalHeight = totalHeight + child.Size.Y.Offset + 10
            end
        end
        ContentInner.Size = UDim2.new(1, -20, 0, totalHeight)
        Content.CanvasSize = UDim2.new(0, 0, 0, totalHeight + 20)
    end
    
    -- Add spacing between panels
    local function addSpacing()
        local panels = {}
        for _, child in ipairs(ContentInner:GetChildren()) do
            if child:IsA("Frame") and child.Name == "Panels" then
                table.insert(panels, child)
            end
        end
        for i, panel in ipairs(panels) do
            if i > 1 then
                local prev = panels[i - 1]
                panel.Position = UDim2.new(0, 0, 0, prev.Position.Y.Offset + prev.Size.Y.Offset + 10)
            else
                panel.Position = UDim2.new(0, 0, 0, 0)
            end
        end
        updateContentSize()
    end
    
    addSpacing()

    local function animateObject(obj, properties, duration, easingStyle, easingDirection)
        if not Settings.AnimationsEnabled then
            for prop, value in pairs(properties) do
                obj[prop] = value
            end
            return
        end
        
        local tweenInfo = TweenInfo.new(
            duration or 0.3,
            easingStyle or Enum.EasingStyle.Quad,
            easingDirection or Enum.EasingDirection.Out
        )
        local tween = TweenService:Create(obj, tweenInfo, properties)
        tween:Play()
        return tween
    end
    
    local function tweenTransparency(obj, target, duration)
        return animateObject(obj, {BackgroundTransparency = target}, duration or 0.2, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    end

    -- ============================================
    -- NOTIFICATION SCANNER
    -- ============================================
    
    local NotificationCache = {}
    local NotificationCount = 0
    
    local function scanNotifications()
        -- Look for notification templates
        local template = PlayerGui:FindFirstChild("Notification")
        if template then
            template = template:FindFirstChild("Notification")
            if template then
                local desc = template:GetDescendants()
                for _, obj in ipairs(desc) do
                    if obj:IsA("TextLabel") and obj.Name == "Template" then
                        -- Check if this notification contains redeemable text
                        if obj.Text and obj.Text ~= "" then
                            if not NotificationCache[obj] then
                                NotificationCache[obj] = true
                                NotificationCount = NotificationCount + 1
                                NotifCount.Text = NotificationCount .. " caught"
                                
                                local text = obj.Text
                                if isRedeemable(text) then
                                    StatusLabel.Text = "🎯 Notification caught: " .. text
                                    StatusLabel.TextColor3 = Colors.GOLD
                                    
                                    -- Check if armed or auto redeem
                                    if Settings.AutoRedeem or Settings.Armed then
                                        local words = splitWords(text)
                                        -- Process the notification
                                        if #words > 0 then
                                            -- If armed, disable after use
                                            if Settings.Armed then
                                                Settings.Armed = false
                                                ArmBtn.Text = "Arm"
                                                ArmBtn.BackgroundColor3 = Colors.INNER
                                            end
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end
    end
    
    -- Scan periodically
    local function startScanner()
        task.spawn(function()
            while Dashboard.Parent do
                scanNotifications()
                task.wait(1)
            end
        end)
    end
    
    startScanner()

    local dragging = false

    local startPos = nil
    
    local function startDrag(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or 
           input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = Main.Position
        end
    end
    
    local function updateDrag(input)
        if dragging then
            local delta = input.Position - dragStart
            Main.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end
    
    local function endDrag()
        dragging = false
    end
    
    TitleFrame.InputBegan:Connect(startDrag)
    TitleFrame.InputChanged:Connect(updateDrag)
    TitleFrame.InputEnded:Connect(endDrag)

                AskBtn.MouseButton1Click:Fire()

                ManualBtn.MouseButton1Click:Fire()
            end
        end
    end
